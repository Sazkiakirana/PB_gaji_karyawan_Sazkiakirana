  function baca(key, cadangan) {
    try {
      const s = localStorage.getItem(key);
      if (s) return JSON.parse(s);
    } catch (e) {}
    return cadangan;
  }
  function tulis(key, nilai) {
    try { localStorage.setItem(key, JSON.stringify(nilai)); } catch (e) {}
  }

  /* =====================================================
     KEAMANAN PASSWORD
     Password tidak disimpan apa adanya, melainkan di-hash (SHA-256 + salt).
     ===================================================== */
  function acak() {
    const a = new Uint8Array(12);
    crypto.getRandomValues(a);
    return Array.from(a, b => b.toString(16).padStart(2, "0")).join("");
  }
  
  async function hashPw(pw, salt) {
    const teks = salt + ":" + pw;
    if (window.crypto && crypto.subtle) {
      const buf = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(teks));
      return Array.from(new Uint8Array(buf), b => b.toString(16).padStart(2, "0")).join("");
    }
    // cadangan bila browser tidak mendukung crypto.subtle
    let h = 5381;
    for (const c of teks) h = ((h << 5) + h + c.charCodeAt(0)) >>> 0;
    return "f" + h.toString(16);
  }
  
  async function buatAkun(username, nama, role, password) {
    const salt = acak();
    return { username, nama, role, salt, hash: await hashPw(password, salt) };
  }

  function sisaKunci() {
    const k = baca(KEY_KUNCI, { n: 0, sampai: 0 });
    return Math.max(0, Math.ceil((k.sampai - Date.now()) / 1000));
  }

      return;
    }
  
    const username = $("lUser").value.trim().toLowerCase();
    const password = $("lPass").value;
    const akun = pengguna.find(u => u.username === username);
    const benar = akun && (await hashPw(password, akun.salt)) === akun.hash;
  
    if (!benar) {
      const k = baca(KEY_KUNCI, { n: 0, sampai: 0 });
      k.n += 1;
      if (k.n >= 5) {
        k.n = 0;
        k.sampai = Date.now() + 30000;

      } else {

  function hitung(k) {
    const baris = ATURAN.gajiPokok.find(r => k.masa >= r[0] && k.masa <= r[1]);
    const pokok = baris ? baris[2] : 0;
    const tMasa = k.masa >= ATURAN.batasMasaKerja
      ? ATURAN.tunjanganMasaKerjaBesar
      : ATURAN.tunjanganMasaKerjaKecil;
    const tAnak = k.anak >= ATURAN.batasAnak ? pokok * ATURAN.persenAnak : 0;
    const freeme = pokok * ATURAN.persenFreeme;
    const ppl = pokok * ATURAN.persenPPL;
    const bersih = pokok + tMasa + tAnak + freeme + ppl;
    return { pokok, tMasa, tAnak, freeme, ppl, bersih };
  }
  
  const rupiah = n => "Rp " + Math.round(n).toLocaleString("id-ID");

    return String(s).replace(/[&<>"']/g, c => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));
  }

  function gambarDashboard() {
    const hasil = karyawan.map(k => ({ k, g: hitung(k) }));
    const n = hasil.length;
    const total = hasil.reduce((s, x) => s + x.g.bersih, 0);
    const tertinggi = n ? Math.max(...hasil.map(x => x.g.bersih)) : 0;
  
    $("statTotal").textContent = rupiah(total);
    $("statJumlah").textContent = n;
    $("statRata").textContent = rupiah(n ? total / n : 0);
    $("statMax").textContent = rupiah(tertinggi);
  
    const komp = [
      ["Gaji pokok", "--c-pokok", hasil.reduce((s, x) => s + x.g.pokok, 0)],
      ["Tunjangan masa kerja", "--c-masa", hasil.reduce((s, x) => s + x.g.tMasa, 0)],
      ["Tunjangan anak", "--c-anak", hasil.reduce((s, x) => s + x.g.tAnak, 0)],
      ["Freeme 3%", "--c-freeme", hasil.reduce((s, x) => s + x.g.freeme, 0)],
      ["PPL 3%", "--c-ppl", hasil.reduce((s, x) => s + x.g.ppl, 0)],
    ];
    let mulai = 0;
    const stops = [];
    komp.forEach(([, warna, nilai]) => {
      const persen = total ? (nilai / total) * 100 : 0;
      stops.push(`var(${warna}) ${mulai}% ${mulai + persen}%`);
      mulai += persen;
    });
    $("donut").style.background = total ? `conic-gradient(${stops.join(",")})` : "var(--line)";
    $("donutCenter").textContent = n ? "100%" : "-";
    $("legend").innerHTML = komp.map(([nama, warna, nilai]) => `
      <li><i style="background:var(${warna})"></i>${nama}<b>${total ? ((nilai / total) * 100).toFixed(1) : 0}%</b></li>`
    ).join("");
  
    const pend = ["SD", "SMP", "SMA/SMK", "S1", "S2"];
    gambarBar($("barsPendidikan"), pend.map(p => [p, karyawan.filter(k => k.pendidikan === p).length]));
  
    const kel = [
      ["5 – 10 thn", k => k.masa >= 5 && k.masa <= 10],
      ["11 – 15 thn", k => k.masa >= 11 && k.masa <= 15],
      ["16 – 32 thn", k => k.masa >= 16 && k.masa <= 32],
    ];
    gambarBar($("barsMasa"), kel.map(([nama, fn]) => [nama, karyawan.filter(fn).length]));
  
    const top = [...hasil].sort((a, b) => b.g.bersih - a.g.bersih).slice(0, 5);
    $("topList").innerHTML = top.length
      ? top.map(x => `
          <li>
            <div>${esc(x.k.nama)}<small>${esc(x.k.pendidikan)} · ${x.k.masa} tahun</small></div>
            <strong>${rupiah(x.g.bersih)}</strong>
          </li>`).join("")
      : "<li>Belum ada data</li>";
  }
  
  function gambarBar(el, pasangan) {
    const maks = Math.max(1, ...pasangan.map(p => p[1]));
    el.innerHTML = pasangan.map(([nama, jml]) => `
      <div class="bar-row">
        <span>${nama}</span>
        <div class="bar-track"><div class="bar-fill" style="width:${(jml / maks) * 100}%"></div></div>
        <b>${jml}</b>
      </div>`).join("");
  }
